Google Apps Script DB 설정

선생님의 **구글 스프레드시트**와 웹앱을 연결하여 전체 학급의 아침 공부 기록을 데이터베이스에 저장하세요.

📋 선생님용 Google Apps Script 코드

구글 시트 > 확장 프로그램 > Apps Script에 아래 코드를 복사하여 붙여넣고 [웹 앱으로 배포]하세요.

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    rows.push({
      id: data[i][0],
      nickname: data[i][1],
      resolution: data[i][2],
      photo: data[i][3],
      date: data[i][4],
      timestamp: data[i][5],
      cheers: data[i][6] || 0
    });
  }
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Nickname", "Resolution", "Photo", "Date", "Timestamp", "Cheers"]);
  }
  var param = JSON.parse(e.postData.contents);
  if (param.action === "add") {
    sheet.appendRow([
      param.id,
      param.nickname,
      param.resolution,
      param.photo,
      param.date,
      param.timestamp,
      0
    ]);
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
